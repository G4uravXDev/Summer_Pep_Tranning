# Problem Name: Distance from the source (Bellman-Ford Algorithm)
# Link : https://www.geeksforgeeks.org/problems/distance-from-the-source-bellman-ford-algorithm/1
# Topic : Graph (Bellman-Ford Algorithm)


class Solution:
    def bellmanFord(self, V, edges, src):
        INF = 10**8
        dist = [INF] * V
        dist[src] = 0

        for _ in range(V - 1):
            updated = False
            for u, v, w in edges:
                if dist[u] != INF and dist[u] + w < dist[v]:
                    dist[v] = dist[u] + w
                    updated = True
            if not updated:
                break

        for u, v, w in edges:
            if dist[u] != INF and dist[u] + w < dist[v]:
                return [-1]

        return dist
