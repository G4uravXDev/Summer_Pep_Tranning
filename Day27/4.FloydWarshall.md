# Problem Name: Implementing Floyd Warshall
# Link : https://www.geeksforgeeks.org/problems/implementing-floyd-warshall2042/1
# Topic : Graph (Floyd Warshall Algorithm)


class Solution:
    def floydWarshall(self, dist):
        n = len(dist)
        INF = 10**8

        for via in range(n):
            for i in range(n):
                if dist[i][via] == INF:
                    continue
                for j in range(n):
                    if dist[via][j] == INF:
                        continue
                    if dist[i][via] + dist[via][j] < dist[i][j]:
                        dist[i][j] = dist[i][via] + dist[via][j]
        
        return dist
