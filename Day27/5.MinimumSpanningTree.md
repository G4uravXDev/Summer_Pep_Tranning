# Problem Name: Minimum Spanning Tree
# Link : https://www.geeksforgeeks.org/problems/minimum-spanning-tree/1
# Topic : Graph (Prim's Algorithm)

import heapq

class Solution:
    def spanningTree(self, V, edges):
        graph = [[] for _ in range(V)]

        for u, v, w in edges:
            graph[u].append((v, w))
            graph[v].append((u, w))

        visited = [False] * V
        pq = [(0, 0)]
        ans = 0

        while pq:
            wt, node = heapq.heappop(pq)

            if visited[node]:
                continue

            visited[node] = True
            ans += wt

            for nei, w in graph[node]:
                if not visited[nei]:
                    heapq.heappush(pq, (w, nei))

        return ans
