# Problem Name: Making A Large Island
# Link : https://leetcode.com/problems/making-a-large-island/
# Topic : Disjoint Set Union (DSU)

class Solution:
    def largestIsland(self, grid):
        n = len(grid)

        directions = [(1, 0), (-1, 0), (0, 1), (0, -1)]

        island_size = {}

        island_id = 2

        def dfs(r, c, idx):
            grid[r][c] = idx
            size = 1

            for dr, dc in directions:
                nr = r + dr
                nc = c + dc

                if 0 <= nr < n and 0 <= nc < n and grid[nr][nc] == 1:
                    size += dfs(nr, nc, idx)

            return size

        for i in range(n):
            for j in range(n):
                if grid[i][j] == 1:
                    island_size[island_id] = dfs(i, j, island_id)
                    island_id += 1
        
        ans = max(island_size.values(), default=0)
    
        for i in range(n):
            for j in range(n):
                if grid[i][j] == 0:
                    seen = set()
                    current = 1

                    for dr, dc in directions:
                        ni = i + dr
                        nj = j + dc

                        if 0 <= ni < n and 0 <= nj < n:
                            idx = grid[ni][nj]
                            if idx > 1 and idx not in seen:
                                seen.add(idx)
                                current += island_size[idx]

                    ans = max(ans, current)
        return ans
