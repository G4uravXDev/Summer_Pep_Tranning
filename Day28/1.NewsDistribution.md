# Problem Name: News Distribution
# Link : https://codeforces.com/problemset/problem/1167/C
# Topic : Disjoint Set Union (DSU)



import sys

input = sys.stdin.readline

n, m = map(int, input().split())

parent = list(range(n + 1))

size = [1] * (n + 1)

def find(x):
    while parent[x] != x:
        parent[x] = parent[parent[x]]   
        x = parent[x]
    return x

def union(a, b):
    ra = find(a)
    rb = find(b)

    if ra == rb:
        return

    if size[ra] < size[rb]:
        ra, rb = rb, ra

    parent[rb] = ra
    size[ra] += size[rb]

for _ in range(m):
    arr = list(map(int, input().split()))

    k = arr[0]

    if k == 0:
        continue

    first = arr[1]

    for i in range(2, k + 1):
        union(first, arr[i])

component_size = {}

for i in range(1, n + 1):
    root = find(i)
    component_size[root] = component_size.get(root, 0) + 1

answer = []

for i in range(1, n + 1):
    answer.append(str(component_size[find(i)]))

print(" ".join(answer))
