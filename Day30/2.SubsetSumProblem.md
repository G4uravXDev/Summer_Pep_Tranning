# Problem Name: Subset Sum Problem
# Link : https://www.geeksforgeeks.org/problems/subset-sum-problem-1611555638/1
# Topic : Dynamic Programming (Bottom Up)

class Solution:
    def isSubsetSum(self, arr, sum):
        n = len(arr)
        dp = [[-1] * (sum + 1) for _ in range(n)]

        def solve(index, target):
            if target == 0:
                return True

            if index == n:
                return False

            if dp[index][target] != -1:
                return dp[index][target]

            take = False
            if arr[index] <= target:
                take = solve(index + 1, target - arr[index])

            not_take = solve(index + 1, target)

            dp[index][target] = take or not_take
            return dp[index][target]

        return solve(0, sum)
