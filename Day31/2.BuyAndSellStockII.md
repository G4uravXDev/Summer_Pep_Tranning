# Problem Name: Buy and Sell Stock II
# Link : https://leetcode.com/problems/best-time-to-buy-and-sell-stock-ii/description/
# Topic : Dynamic Programming (Bottom Up)

class Solution:
    def maxProfit(self, prices):
        n = len(prices)

        dp = [[-1] * 2 for _ in range(n)]

        def solve(i, buy):
            if i == n:
                return 0

            if dp[i][buy] != -1:
                return dp[i][buy]

            if buy:
                take = -prices[i] + solve(i + 1, 0)
                skip = solve(i + 1, 1)
                dp[i][buy] = max(take, skip)
            else:
                sell = prices[i] + solve(i + 1, 1)
                hold = solve(i + 1, 0)
                dp[i][buy] = max(sell, hold)

            return dp[i][buy]

        return solve(0, 1)
