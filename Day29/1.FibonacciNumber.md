```python
# Problem Name: Fibonacci Number
# Link : https://leetcode.com/problems/fibonacci-number/description/
# Topic : Dynamic Programming (Bottom Up)

def fib(n: int) -> int:
    if n <= 1:
        return n
    
    a = 0
    b = 1

    for _ in range(2, n + 1):
        c = a + b
        a = b
        b = c

    return b

print(fib(2))
print(fib(3))
print(fib(4))


```
